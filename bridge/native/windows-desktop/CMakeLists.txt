cmake_minimum_required(VERSION 3.24)
project(codeaw_desktop LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The desktop helper targets Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_SKIP_INSTALL_RULES ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)
option(CODEAW_WEBRTC "Build H.264 WebRTC support" ON)
if(CODEAW_WEBRTC)
  add_compile_definitions(MBEDTLS_USER_CONFIG_FILE="${CMAKE_CURRENT_SOURCE_DIR}/mbedtls-config.h")
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git GIT_TAG v3.6.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(mbedtls)
  list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
  set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
  set(NO_TESTS ON CACHE BOOL "" FORCE)
  set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
  FetchContent_Declare(datachannel GIT_REPOSITORY https://github.com/paullouisageneau/libdatachannel.git GIT_TAG v0.24.5 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(datachannel)
endif()
add_executable(codeaw-desktop main.cpp capture.cpp input.cpp service.cpp video.cpp)
target_compile_definitions(codeaw-desktop PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(codeaw-desktop PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(codeaw-desktop PRIVATE nlohmann_json::nlohmann_json d3d11 dxgi windowscodecs ole32 user32 gdi32 advapi32 wtsapi32 userenv sas mfplat mf mfuuid wmcodecdspuuid shlwapi crypt32)
if(CODEAW_WEBRTC)
  target_compile_definitions(codeaw-desktop PRIVATE CODEAW_WEBRTC=1)
  target_link_libraries(codeaw-desktop PRIVATE datachannel-static)
endif()
